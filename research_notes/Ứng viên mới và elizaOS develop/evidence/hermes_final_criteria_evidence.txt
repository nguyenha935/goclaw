## Prototype source: lab-org/__init__.py
"""lab-org: prototype plugin, no core changes.

State lives in $LAB_ORG_DIR (shared by all profiles of one Hermes install):
  org.json    {"admins": [{"platform","user_id"}], "claim_code": str|null,
               "agents": {profile: {title, department, reports_to, kb_scopes}}}
  grants.json [{"viewer": "labgram:1002", "subject": "labgram:1001", "by": ..., "at": ...}]
  kb/<scope>/*.md   shared knowledge files (folder sync)
  audit.jsonl policy decisions
"""
from __future__ import annotations

import json, os, re, time
from pathlib import Path

ENFORCE = os.environ.get("LAB_ORG_ENFORCE", "1") == "1"
PEER_TOOLS = {"honcho_search", "honcho_profile", "honcho_context", "honcho_reasoning", "honcho_conclude"}
ADMIN_ONLY_TOOLS = {"session_search"}
STOP = set("là gì của và có không bao nhiêu mình bạn em anh chị cho với được này đó thì một các những ai nào hãy giúp công ty".split())


def _dir() -> Path:
    return Path(os.environ["LAB_ORG_DIR"])


def _load(name, default):
    try:
        return json.loads((_dir() / name).read_text(encoding="utf-8"))
    except Exception:
        return default


def _save(name, obj):
    (_dir() / name).write_text(json.dumps(obj, ensure_ascii=False, indent=1), encoding="utf-8")


def _audit(**kw):
    with open(_dir() / "audit.jsonl", "a", encoding="utf-8") as f:
        f.write(json.dumps({"t": time.time(), **kw}, ensure_ascii=False) + "\n")


def _profile() -> str:
    from hermes_constants import get_hermes_home
    home = Path(get_hermes_home())
    return home.name if home.parent.name == "profiles" else "default"


def _sender():
    from gateway.session_context import get_session_env
    return (get_session_env("HERMES_SESSION_PLATFORM"), get_session_env("HERMES_SESSION_USER_ID"),
            get_session_env("HERMES_SESSION_USER_NAME"))


def _is_admin(platform, user_id) -> bool:
    return any(a.get("platform") == platform and str(a.get("user_id")) == str(user_id)
               for a in _load("org.json", {}).get("admins", []))


def _granted(platform, viewer, subject) -> bool:
    return any(g["viewer"] == f"{platform}:{viewer}" and g["subject"] == f"{platform}:{subject}"
               for g in _load("grants.json", []))


# ---------------- K1: shared KB -----------------
def _tokens(text):
    return {t for t in re.findall(r"\w+", (text or "").lower()) if len(t) > 1 and t not in STOP}


def _kb_chunks(scopes):
    out = []
    for scope in scopes:
        for f in sorted((_dir() / "kb" / scope).glob("*.md")):
            for para in re.split(r"\n\s*\n", f.read_text(encoding="utf-8")):
                if para.strip():
                    out.append((scope, f.name, para.strip()))
    return out


def _kb_search(query, scopes, k=2):
    q = _tokens(re.sub(r"<memory-context>.*?</memory-context>", "", query or "", flags=re.S))
    scored = sorted(((len(q & _tokens(c)), s, n, c) for s, n, c in _kb_chunks(scopes)), reverse=True)
    return [(s, n, c) for sc, s, n, c in scored[:k] if sc > 0]


def _scopes():
    return _load("org.json", {}).get("agents", {}).get(_profile(), {}).get("kb_scopes", ["company"])


def kb_search_tool(args, **_kw):
    hits = _kb_search(args.get("query", ""), _scopes(), k=int(args.get("limit") or 3))
    return json.dumps({"scopes": _scopes(), "results": [{"scope": s, "file": n, "text": c} for s, n, c in hits]},
                      ensure_ascii=False)

def pre_llm_call(session_id=None, user_message=None, platform=None, sender_id=None, **kw):
    parts = []
    plat, uid, uname = _sender()
    if uid:
        role = "QUẢN TRỊ HỆ THỐNG (xác minh theo user_id)" if _is_admin(plat, uid) else \

        parts.append(f"[lab-org] Người gửi lượt này: platform={plat} user_id={uid} tên hiển thị={uname!r}; vai trò: {role}.")

    _audit(ev="pre_llm_call", profile=_profile(), sender=f"{plat}:{uid}", kb_hits=len(hits))
    return {"context": "\n".join(parts)} if parts else None


def _unwrap(tool_name, args):
    if tool_name == "tool_call" and isinstance(args, dict):
        inner = args.get("name") or ""
        a = args.get("arguments") or {}
        if isinstance(a, str):
            try:
                a = json.loads(a)
            except Exception:
                a = {}
        return inner, a
    return tool_name, args or {}


def pre_tool_call(tool_name=None, args=None, **kw):
    name, a = _unwrap(tool_name, args)
    plat, uid, _ = _sender()
    decision = "allow"
    if not uid:  # CLI / worker without a chat sender: out of scope here
        decision = "allow-no-sender"
    elif name in PEER_TOOLS:
        peer = str(a.get("peer") or "user")
        if peer not in ("user", "ai", uid) and not _is_admin(plat, uid) and not _granted(plat, uid, peer):
            decision = f"block: {plat}:{uid} không được xem peer {peer}"
    elif name in ADMIN_ONLY_TOOLS and not _is_admin(plat, uid):
        decision = f"block: {name} chỉ dành cho quản trị"
    _audit(ev="pre_tool_call", tool=tool_name, inner=name, args=a, sender=f"{plat}:{uid}", decision=decision,
           enforce=ENFORCE)
    if ENFORCE and decision.startswith("block"):
        return {"action": "block", "message": "Từ chối theo chính sách quyền riêng tư: " + decision[7:]}
    return None


async def pre_gateway_dispatch(event=None, gateway=None, **kw):
    text = (getattr(event, "text", "") or "").strip()
    if not text.startswith(("/grant", "/revoke", "/claim_admin")):
        return None
    src = event.source
    plat = src.platform.value if src.platform else "?"
    uid = str(src.user_id)
    parts = text.split()
    org = _load("org.json", {})
    reply = None
    if parts[0] == "/claim_admin":
        code = org.get("claim_code")
        if code and len(parts) == 2 and parts[1] == code:
            org.setdefault("admins", []).append({"platform": plat, "user_id": uid, "bound_at": time.time(),
                                                 "display_name_at_bind": src.user_name})
            org["claim_code"] = None
            _save("org.json", org)
            reply = f"✅ Đã gắn {plat}:{uid} làm quản trị hệ thống."
        else:
            reply = "⛔ Mã không hợp lệ hoặc đã dùng."
    elif not _is_admin(plat, uid):
        reply = f"⛔ {parts[0]} chỉ dành cho quản trị hệ thống. ({plat}:{uid} không có quyền)"
    elif len(parts) == 3:
        grants = _load("grants.json", [])
        g = {"viewer": f"{plat}:{parts[1]}", "subject": f"{plat}:{parts[2]}"}
        grants = [x for x in grants if not (x["viewer"] == g["viewer"] and x["subject"] == g["subject"])]
        if parts[0] == "/grant":
            grants.append({**g, "by": f"{plat}:{uid}", "at": time.time()})
        _save("grants.json", grants)
        reply = f"✅ {parts[0][1:]}: {g['viewer']} → {g['subject']}"
    else:
        reply = "Cú pháp: /grant <viewer_user_id> <subject_user_id>"
    _audit(ev="command", cmd=parts[0], sender=f"{plat}:{uid}", name=src.user_name, reply=reply)
    adapter = gateway._delivery_adapter_for(src) if gateway is not None else None
    if adapter is not None:
        await adapter.send(src.chat_id, reply)
    return {"action": "skip", "reason": "lab-org command"}


def _directory(_info=None) -> str:
    agents = _load("org.json", {}).get("agents", {})
    if not agents:
        return ""
    lines = ["## Sơ đồ phòng ban (lab-org)", "Dùng đúng tên profile khi giao việc bằng kanban_create(assignee=...)."]
    for name, a in agents.items():
        rep = f", báo cáo cho {a['reports_to']}" if a.get("reports_to") else ""
        lines.append(f"- {name}: {a.get('title')} – phòng {a.get('department')}{rep}. {a.get('duties', '')}")
    return "\n".join(lines)

## Prototype source: lab-gram/__init__.py
"""Fake messaging platform 'labgram' for lab runs (modelled on tests/e2e/core/chaos/_gateway_fake_platform.py).

Inbound: JSON files dropped into ``platforms.labgram.extra.inbox`` ->
build_source() -> MessageEvent -> handle_message(), i.e. the same path every third-party adapter takes.
Outbound: every send/edit/complete is appended to ``extra.outbox`` (JSON lines).
"""
from __future__ import annotations

import asyncio, json, os, time
from pathlib import Path
from typing import Any, Dict, Optional

from gateway.config import Platform, PlatformConfig
from gateway.platforms.base import BasePlatformAdapter, SendResult
from gateway.platforms.event import MessageEvent, MessageType

NAME = "labgram"


class LabGramAdapter(BasePlatformAdapter):
    def __init__(self, config: PlatformConfig):
        super().__init__(config=config, platform=Platform(NAME))
        extra = getattr(config, "extra", None) or {}
        self._inbox = Path(extra.get("inbox") or os.environ["LABGRAM_INBOX"])
        self._outbox = Path(extra.get("outbox") or os.environ["LABGRAM_OUTBOX"])
        self._task: Optional[asyncio.Task] = None
        self._seq = 0

    def _emit(self, payload: Dict[str, Any]) -> None:
        payload["t"] = time.time()
        with open(self._outbox, "a", encoding="utf-8") as f:
            f.write(json.dumps(payload, ensure_ascii=False) + "\n")

    async def _poll(self) -> None:
        while True:
            try:
                for p in sorted(self._inbox.glob("*.json")):
                    cmd = json.loads(p.read_text(encoding="utf-8"))
                    p.rename(p.with_suffix(".done"))
                    await self._inbound(cmd)
            except Exception as exc:  # keep polling
                self._emit({"ev": "error", "error": repr(exc)})
            await asyncio.sleep(0.2)

    async def _inbound(self, cmd: Dict[str, Any]) -> None:
        msg_id = str(cmd.get("id"))
        source = self.build_source(
            chat_id=str(cmd["chat_id"]), chat_name=cmd.get("chat_name"), chat_type=cmd.get("chat_type", "dm"),
            user_id=str(cmd["user_id"]), user_name=cmd.get("user_name"), message_id=msg_id)
        event = MessageEvent(text=str(cmd.get("text") or ""), message_type=MessageType.TEXT,
                             source=source, message_id=msg_id)
        self._emit({"ev": "inbound", "id": msg_id, "chat": cmd["chat_id"], "user": cmd["user_id"], "text": cmd.get("text")})
        await self.handle_message(event)

    async def connect(self, *, is_reconnect: bool = False) -> bool:
        self._inbox.mkdir(parents=True, exist_ok=True)
        self._task = asyncio.create_task(self._poll(), name="labgram-inbox")
        self._mark_connected()
        self._emit({"ev": "ready", "pid": os.getpid(), "inbox": str(self._inbox)})
        return True

    async def disconnect(self) -> None:
        self._mark_disconnected()
        if self._task:
            self._task.cancel()

    async def send(self, chat_id: str, content: str, reply_to: Optional[str] = None,
                   metadata: Optional[Dict[str, Any]] = None) -> SendResult:
        self._seq += 1
        mid = f"out-{self._seq}"
        self._emit({"ev": "send", "chat": str(chat_id), "text": content, "id": mid})
        return SendResult(success=True, message_id=mid)

    async def edit_message(self, chat_id: str, message_id: str, content: str, **_kw: Any) -> SendResult:
        self._emit({"ev": "send", "chat": str(chat_id), "text": content, "id": message_id, "edit": True})
        return SendResult(success=True, message_id=message_id)

    async def on_processing_complete(self, event: MessageEvent, outcome: Any) -> None:
        self._emit({"ev": "complete", "id": event.message_id, "outcome": str(outcome)})

    async def get_chat_info(self, chat_id: str) -> Dict[str, Any]:
        return {"name": chat_id, "type": "group" if str(chat_id).startswith("-") else "dm"}


def register(ctx) -> None:
    ctx.register_platform(
        name=NAME, label="Lab Gram", adapter_factory=LabGramAdapter, check_fn=lambda: True,
        validate_config=lambda _cfg: True, is_connected=lambda _cfg: True,
        allow_all_env="LABGRAM_ALLOW_ALL_USERS", allowed_users_env="LABGRAM_ALLOWED_USERS",
    )
